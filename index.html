2. Проверить полученные значения хэшей на веб-ресурсах

Команды для Windows (CMD):

Дополнительное задание: Python скрипт для расчета хэшей

Программа на Python для вычисления хэш-значений:

import hashlib

def calculate_file_hash(file_path, algorithm='sha256'):
    """Вычисляет хэш файла"""
    hash_object = hashlib.new(algorithm)
    try:
        with open(file_path, 'rb') as f:
            for chunk in iter(lambda: f.read(4096), b''):
                hash_object.update(chunk)
        return hash_object.hexdigest()
    except Exception as e:
        return f"Ошибка: {e}"

def calculate_string_hash(text, algorithm='sha256'):
    """Вычисляет хэш строки"""
    try:
        hash_object = hashlib.new(algorithm)
        hash_object.update(text.encode('utf-8'))
        return hash_object.hexdigest()
    except Exception as e:
        return f"Ошибка: {e}"

# Пример использования
if __name__ == "__main__":
    print("1. Вычислить хэш файла")
    print("2. Вычислить хэш строки")
    choice = input("Выберите действие (1/2): ")
    
    if choice == '1':
        file_path = input("Введите путь к файлу: ")
        algorithm = input("Введите алгоритм (sha256 по умолчанию): ") or 'sha256'
        result = calculate_file_hash(file_path, algorithm)
        print(f"Хэш ({algorithm}) файла: {result}")
    elif choice == '2':
        text = input("Введите текст: ")
        algorithm = input("Введите алгоритм (sha256 по умолчанию): ") or 'sha256'
        result = calculate_string_hash(text, algorithm)
        print(f"Хэш ({algorithm}) строки: {result}")
    else:
        print("Неверный выбор")

Как использовать:

1 Сохраните скрипт как hash_calculator.py

2 Для файлов: python hash_calculator.py → выберите 1 → укажите путь к файлу

3 Для строк: python hash_calculator.py → выберите 2 → введите текст

Поддерживаемые алгоритмы:

sha1, sha224, sha256, sha384, sha512, md5 и другие (зависит от системы)
